     1→-- =============================================================================
     2→-- ALO English Test Platform - AI Agent Employee Addon (ai_agents_addon.sql)
     3→-- =============================================================================
     4→-- Project   : ALO English Test Platform (cPanel production package)
     5→-- Engine    : InnoDB / utf8mb4_unicode_ci
     6→-- Scope     : 10 tables for the AI Agent Employee workforce + seed data.
     7→--             20 AI Agent Employees covering intake, eligibility, setup,
     8→--             proctoring, exam operations, evaluation, scoring, feedback,
     9→--             reports, CRM sync, voucher/payment, content authoring,
    10→--             compliance, audit, and student support.
    11→-- Usage     : (a) Import standalone into a database already running install.sql,
    12→--             OR (b) the same content is appended to the end of install.sql
    13→--             so that a single install.sql import creates everything.
    14→-- Compliance: Original content only. No official exam items reproduced.
    15→--             All public-facing labels use ONLY approved wording:
    16→--               "ALO Evaluation System", "Digital Practice Session",
    17→--               "ALO Practice Examiner", "Practice Interview Coach",
    18→--               "Automated Practice Feedback", "Smart Score Report",
    19→--               "ALO Estimated Score Engine".
    20→--             No claim of "AI teacher", "bot", "robot", "fake human",
    21→--             "real examiner", "official examiner", or "official score"
    22→--             anywhere in the public-facing labels.
    23→--             internal_label uses "Agent" terminology for admin-only display.
    24→-- Disclosure: "Some practice tests, scoring, feedback, and interview
    25→--             simulations may be supported by automated digital evaluation
    26→--             tools. This is a practice platform only, not an official
    27→--             exam result."
    28→-- =============================================================================

    50→-- 1. exam_ai_agents  (AI Agent Employee master register)
    51→-- -----------------------------------------------------------------------------
    52→CREATE TABLE exam_ai_agents (
    53→    id                      BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    54→    agent_code              VARCHAR(48)  NOT NULL,
    55→    agent_name              VARCHAR(150) NOT NULL,
    56→    department              VARCHAR(80)  NOT NULL,
    57→    responsibility          VARCHAR(500) NOT NULL,
    58→    exam_scope              JSON         NOT NULL,
    59→    is_active               TINYINT(1)   NOT NULL DEFAULT 1,
    60→    public_label            VARCHAR(120) NOT NULL,
    61→    internal_label          VARCHAR(120) NOT NULL,
    62→    can_auto_execute        TINYINT(1)   NOT NULL DEFAULT 0,
    63→    requires_admin_approval TINYINT(1)   NOT NULL DEFAULT 1,
    64→    created_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    65→    updated_at              DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    66→    PRIMARY KEY (id),
    67→    UNIQUE KEY uq_ai_agents_code (agent_code),
    68→    KEY idx_ai_agents_code (agent_code),
    69→    KEY idx_ai_agents_active (is_active),
    70→    KEY idx_ai_agents_department (department)
    71→) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='AI Agent Employee master register (admin-only "Agent" terminology)';

    76→CREATE TABLE exam_ai_agent_tasks (

    80→    attempt_id     BIGINT UNSIGNED NULL,

    84→    result         JSON         NULL,
    85→    started_at     DATETIME     NULL,
    86→    completed_at   DATETIME     NULL,
    87→    error_message  TEXT         NULL,
    88→    created_at     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    89→    updated_at     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    90→    PRIMARY KEY (id),
    91→    UNIQUE KEY uq_ai_tasks_code (task_code),
    92→    KEY idx_ai_tasks_agent (agent_id),
    93→    KEY idx_ai_tasks_status (status),
    94→    KEY idx_ai_tasks_attempt (attempt_id),
    95→    CONSTRAINT fk_ai_tasks_agent   FOREIGN KEY (agent_id)   REFERENCES exam_ai_agents(id)        ON DELETE CASCADE,
